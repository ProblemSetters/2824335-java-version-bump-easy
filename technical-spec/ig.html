package com.versions;

import com.versions.Models.Change;
import com.versions.Models.ChangeKind;
import com.versions.Models.CommandResult;
import com.versions.Models.Reason;
import com.versions.Models.Release;
import com.versions.Models.State;
import com.versions.Models.Version;

public final class Commands {
    private Commands() {}

    /// Find a release by its version number.
    private static Release findRelease(State state, String version) {
        for (Release r : state.releases) {
            if (r.version.toString().equals(version)) {
                return r;
            }
        }
        return null;
    }

    /// How big a change of this kind is, from patch up to major.
    private static int typeRank(ChangeKind kind) {
        switch (kind) {
            case MAJOR: return 3;
            case MINOR: return 2;
            case PATCH: return 1;
        }
        return 1;
    }

    /// The version a change of this kind produces from the current one.
    private static Version bump(Version current, ChangeKind kind) {
        Version next = new Version(current.major, current.minor, current.patch);
        if (kind == ChangeKind.MAJOR) {
            next.major += 1;
            next.minor = 0;
            next.patch = 0;
        } else if (kind == ChangeKind.MINOR) {
            next.minor += 1;
            next.patch = 0;
        } else {
            next.patch += 1;
        }
        return next;
    }

    /// True when a release other than the given version has not been pulled.
    private static boolean hasOtherLiveRelease(State state, String version) {
        for (Release release : state.releases) {
            if (!release.yanked && !release.version.toString().equals(version)) {
                return true;
            }
        }
        return false;
    }

    /// Build a successful result carrying the command's output message.
    private static CommandResult success(String message) {
        CommandResult result = new CommandResult();
        result.ok = true;
        result.message = message;
        return result;
    }

    /// Build a failed result carrying the reason code.
    private static CommandResult failure(Reason reason) {
        CommandResult result = new CommandResult();
        result.reason = reason;
        return result;
    }

    /// Note a change for the next release.
    public static CommandResult change(State state, String type, String note) {
        if (!"major".equals(type) && !"minor".equals(type) && !"patch".equals(type)) {
            return failure(Reason.INVALID_TYPE);
        }
        if (note.isEmpty()) {
            return failure(Reason.EMPTY_NOTE);
        }

        Change noted = new Change(ChangeKind.parseChangeKind(type), note);
        state.pending.add(noted);

        return success("Noted " + type + ": " + note + " (" + state.pending.size() + " pending)");
    }

    /// Turn the pending changes into a new version.
    public static CommandResult release(State state) {
        if (state.pending.isEmpty()) {
            return failure(Reason.NOTHING_TO_RELEASE);
        }

        ChangeKind bumpKind = ChangeKind.PATCH;
        for (Change change : state.pending) {
            if (typeRank(change.type) > typeRank(bumpKind)) {
                bumpKind = change.type;
            }
        }

        Release fresh = new Release();
        fresh.version = bump(state.version, bumpKind);
        for (Change change : state.pending) {
            fresh.notes.add(change.note);
        }
        fresh.yanked = false;

        int count = state.pending.size();
        state.pending.clear();
        state.releases.add(0, fresh);
        state.version = fresh.version;

        return success("Released " + fresh.version + " (" + count + " changes)");
    }

    /// Pull a release.
    public static CommandResult yank(State state, String version) {
        Release target = findRelease(state, version);
        if (target == null) {
            return failure(Reason.VERSION_NOT_FOUND);
        }
        if (target.yanked) {
            return failure(Reason.ALREADY_YANKED);
        }
        boolean wasCurrent = state.version.toString().equals(version);
        if (wasCurrent && !hasOtherLiveRelease(state, version)) {
            return failure(Reason.NO_OTHER_VERSION);
        }

        target.yanked = true;
        if (wasCurrent) {
            for (Release release : state.releases) {
                if (!release.yanked) {
                    state.version = release.version;
                    break;
                }
            }
        }

        return success("Yanked " + version + "; current version " + state.version);
    }
}

package com.versions;

import com.versions.Models.Change;
import com.versions.Models.ChangeKind;
import com.versions.Models.Release;
import com.versions.Models.State;
import com.versions.Models.Version;

public final class Parser {
    private Parser() {}

    /// Read a version number from its text form.
    public static Version parseVersionText(String text) {
        int first = text.indexOf('.');
        int second = text.indexOf('.', first + 1);
        Version version = new Version();
        version.major = Integer.parseInt(text.substring(0, first));
        version.minor = Integer.parseInt(text.substring(first + 1, second));
        version.patch = Integer.parseInt(text.substring(second + 1));
        return version;
    }

    private static class JsonParser {
        private final String json;
        private int pos;

        public JsonParser(String json) {
            this.json = json != null ? json : "";
            this.pos = 0;
        }

        /// Parse the Version Bump state document into the in-memory State.
        public State parseVersionsDoc() {
            State state = new State();
            skipWs();
            expect('{');
            while (true) {
                skipWs();
                if (peek() == '}') {
                    pos++;
                    break;
                }
                String key = parseString();
                skipWs();
                expect(':');
                skipWs();
                if ("version".equals(key)) {
                    state.version = parseVersionText(parseString());
                } else if ("pending".equals(key)) {
                    parseArray(() -> state.pending.add(parseChangeObject()));
                } else if ("releases".equals(key)) {
                    parseArray(() -> state.releases.add(parseReleaseObject()));
                }
                skipWs();
                if (peek() == ',') {
                    pos++;
                }
            }
            return state;
        }

        /// Run parseElement once per item in a JSON array.
        private void parseArray(Runnable parseElement) {
            expect('[');
            skipWs();
            if (peek() == ']') {
                pos++;
                return;
            }
            while (true) {
                parseElement.run();
                skipWs();
                if (peek() == ',') {
                    pos++;
                    skipWs();
                    continue;
                }
                expect(']');
                break;
            }
        }

        /// Parse a single pending change object.
        private Change parseChangeObject() {
            Change change = new Change();
            skipWs();
            expect('{');
            while (true) {
                skipWs();
                if (peek() == '}') {
                    pos++;
                    break;
                }
                String key = parseString();
                skipWs();
                expect(':');
                skipWs();
                if ("type".equals(key)) {
                    change.type = ChangeKind.parseChangeKind(parseString());
                } else if ("note".equals(key)) {
                    change.note = parseString();
                }
                skipWs();
                if (peek() == ',') {
                    pos++;
                }
            }
            return change;
        }

        /// Parse a single release object.
        private Release parseReleaseObject() {
            Release release = new Release();
            skipWs();
            expect('{');
            while (true) {
                skipWs();
                if (peek() == '}') {
                    pos++;
                    break;
                }
                String key = parseString();
                skipWs();
                expect(':');
                skipWs();
                if ("version".equals(key)) {
                    release.version = parseVersionText(parseString());
                } else if ("notes".equals(key)) {
                    parseArray(() -> release.notes.add(parseString()));
                } else if ("yanked".equals(key)) {
                    release.yanked = parseBool();
                }
                skipWs();
                if (peek() == ',') {
                    pos++;
                }
            }
            return release;
        }

        private char peek() {
            return pos < json.length() ? json.charAt(pos) : '\0';
        }

        private void expect(char c) {
            if (peek() != c) {
                throw new RuntimeException("JSON parse error: expected '" + c + "' at position " + pos);
            }
            pos++;
        }

        private void skipWs() {
            while (pos < json.length() && Character.isWhitespace(json.charAt(pos))) {
                pos++;
            }
        }

        private String parseString() {
            expect('"');
            StringBuilder value = new StringBuilder();
            while (pos < json.length() && json.charAt(pos) != '"') {
                if (json.charAt(pos) == '\\' && pos + 1 < json.length()) {
                    pos++;
                    char escaped = json.charAt(pos++);
                    switch (escaped) {
                        case '"':  value.append('"');  break;
                        case '\\': value.append('\\'); break;
                        case '/':  value.append('/');  break;
                        case 'n':  value.append('\n'); break;
                        case 'r':  value.append('\r'); break;
                        case 't':  value.append('\t'); break;
                        default:   value.append(escaped);
                    }
                } else {
                    value.append(json.charAt(pos++));
                }
            }
            expect('"');
            return value.toString();
        }

        private boolean parseBool() {
            if (json.startsWith("true", pos)) {
                pos += 4;
                return true;
            }
            if (json.startsWith("false", pos)) {
                pos += 5;
                return false;
            }
            throw new RuntimeException("JSON parse error: expected boolean at position " + pos);
        }
    }

    /// Parse the Version Bump state JSON into the in-memory State.
    public static State parseVersionsJson(String doc) {
        JsonParser parser = new JsonParser(doc);
        return parser.parseVersionsDoc();
    }
}
